\newread\synthetictwentyusfile%

\newcommand{\IfNonemptySyntheticTwentyUsFile}[3]{%
    \IfFileExists{#1}{%
        \openin\synthetictwentyusfile=#1
        \ifeof\synthetictwentyusfile%
            \closein\synthetictwentyusfile%
            #3%
        \else
            \closein\synthetictwentyusfile%
            #2%
        \fi
    }{%
        #3%
    }%
}

\newcommand{\SyntheticTwentyUsPlot}[3]{%
    \IfNonemptySyntheticTwentyUsFile{#3}{%
        \addplot+[#1]
        table [#2, col sep=comma] {#3};
    }{}%
}

\newcommand{\SyntheticTwentyUsSchedsimPlot}[1]{%
    \SyntheticTwentyUsPlot{mark=none, solid, cLightBlue}{x expr=\thisrow{Req/time_unit} * 50000, y expr=\thisrow{99th (us)} / 50}{data/schedsim/single_queue/FIFO_1000_20_80_#1.csv}%
}

\newcommand{\SyntheticTwentyUsYAxisStyle}{%
    scaled y ticks=false,
    ytick={200, 400, 600, 800, 1000},
    yticklabel style={
        /pgf/number format/fixed,
        /pgf/number format/precision=0,
    }%
}
