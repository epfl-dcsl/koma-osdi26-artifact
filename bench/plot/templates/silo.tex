\newread\silofile%

\newcommand{\IfNonemptySiloFile}[3]{%
    \IfFileExists{#1}{%
        \openin\silofile=#1
        \ifeof\silofile%
            \closein\silofile%
            #3%
        \else
            \closein\silofile%
            #2%
        \fi
    }{%
        #3%
    }%
}

\newcommand{\SiloPlot}[3]{%
    \IfNonemptySiloFile{#3}{%
        \addplot+[#1]
        table [#2, col sep=comma] {#3};
    }{}%
}
